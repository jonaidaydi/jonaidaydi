from pathlib import Path
import re

path = Path("/mnt/data/README.md")
text = path.read_text(encoding="utf-8")

# Extract GitHub section
pattern = re.compile(r"\n---\n\n## 📊 GitHub\n.*?(?=\n---\n\n## 🌍 Find me)", re.S)
m = pattern.search(text)
if not m:
    raise RuntimeError("GitHub section not found.")

github_section = m.group(0)

# Remove GitHub section from its current location
text = text[:m.start()] + "\n" + text[m.end():]

# Place the GitHub section directly before About me
about_marker = "\n---\n\n## 🚀 About me"
if about_marker not in text:
    raise RuntimeError("About me section not found.")

text = text.replace(
    about_marker,
    github_section + "\n\n---\n\n## 🚀 About me",
    1
)

# Remove Luxembourg closing line
text = re.sub(
    r'\n<p align="center">\s*<b>🇱🇺 Moien from Luxembourg\.</b>\s*</p>\s*$',
    "\n",
    text,
    flags=re.S
)

path.write_text(text, encoding="utf-8")
print(f"Updated {path}")
